# ==============================================================================
# Linux Data Guardian — Secure Sudo Installation Steps
# ==============================================================================
# Run these commands manually in your terminal when ready.
# They install the root-owned privileged helper and targeted sudoers permissions,
# and remove the old unsafe APT hook.
# ==============================================================================

# 1. Install the root-owned privileged helper script
sudo mkdir -p /usr/local/libexec
sudo cp "/home/jagdish/Desktop/System Files/Linux Data Guardian/scripts/data-guardian-helper" /usr/local/libexec/data-guardian-helper
sudo chown root:root /usr/local/libexec/data-guardian-helper
sudo chmod 0755 /usr/local/libexec/data-guardian-helper

# 2. Install targeted sudoers configuration (allows ONLY data-guardian-helper)
sudo cp "/home/jagdish/Desktop/System Files/Linux Data Guardian/config/sudoers-data-guardian" /etc/sudoers.d/linux-data-guardian
sudo chown root:root /etc/sudoers.d/linux-data-guardian
sudo chmod 0440 /etc/sudoers.d/linux-data-guardian

# 3. Remove the legacy root APT execution hook (replaced by kernel-level UID 42 isolation)
sudo rm -f /etc/apt/apt.conf.d/99data-guardian

# 4. Restart the background daemon service
sudo systemctl restart data-guardian.service

# 5. Verify the helper is functioning cleanly
sudo -n /usr/local/libexec/data-guardian-helper --status
